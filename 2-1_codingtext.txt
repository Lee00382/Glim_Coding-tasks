#include <string>
#include <vector>

using namespace std;

int solution(int n, int k) {
    int lamb_price = 12000;
    int drink_price = 2000;
    
    int free_drinks = n / 10;
    
    int total_cost = (n * lamb_price) + ((k - free_drinks) * (drink_price));
    
    return total_cost;
}
